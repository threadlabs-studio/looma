# ui-disclosure

Progressively enhances a button/content pair into an accessible disclosure.

## Component Reference

<ComponentExamplesApiTabs component="ui-disclosure" />

## Attributes

| Name | Type | Default |
| --- | --- | --- |
| `open` | `boolean` | `false` |
| `default-open` | `boolean` | `false` |
| `disabled` | `boolean` | `false` |

## Properties

| Name | Type |
| --- | --- |
| `open` | `boolean` |
| `defaultOpen` | `boolean` |
| `disabled` | `boolean` |

## Events

| Name | Detail |
| --- | --- |
| `open` | `{ open: true, reason, trigger }` |
| `close` | `{ open: false, reason, trigger }` |

## Slots

Default slot with authored trigger/content children.

## SSR Markup

```html
<ui-disclosure>
  <button type="button" aria-controls="faq-a1">What is light DOM?</button>
  <div id="faq-a1" hidden>Inspectable SSR markup with progressive behavior.</div>
</ui-disclosure>
```

import { h } from "vue";
import { Disclosure } from "@looma/vue";

export const Example = {
  setup() {
    return () =>
      h(Disclosure, { onOpen: (detail: unknown) => console.log(detail) }, {
        default: () => [
          h("button", { type: "button", "aria-controls": "faq-a1" }, "What is light DOM?"),
          h("div", { id: "faq-a1", hidden: true }, "Inspectable SSR markup with progressive behavior.")
        ]
      });
  }
};
